\subsubsection{Authentication \& Authorization}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\textwidth]{graphics/chapter3/system/system.png}
    \caption{Sequence diagram of Authentication \& Authorization}
    \label{fig:authentication_authorization}
\end{figure}

\newpage
\noindent \textbf{Description} \\ 
\textbf{Actors:} Student, Tutor, Admin, System, HCMUT\_SSO, Database

\begin{enumerate}
    \item Users (Student/Tutor/Admin) submit login credentials to System
    \item System authenticates via HCMUT\_SSO
    \item HCMUT\_SSO returns JWT token and role information
    \item System stores session in Database
    \item System redirects users to role-based dashboards
    \item Users request protected resources
    \item System validates token and checks permissions via Database
    \item System grants access with 200 OK response
    \item System detects inactivity and shows timeout warning
    \item User activity triggers session extension in Database
    \item System confirms session renewal to user
\end{enumerate}

\subsubsection{System Integration}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.9\textwidth]{graphics/chapter3/data/system.png}
    \caption{Sequence diagram of HCMUT Data Integration}
    \label{fig:hcmut_data_integration}
\end{figure}

\newpage
\noindent \textbf{Description} \\ 
\textbf{Actors:} Student, Tutor, Admin, System, HCMUT\_SSO, HCMUT\_DATACORE, HCMUT\_LIBRARY, Database

\begin{enumerate}
    \item Student, Tutor, and Admin access the system
    \item System sends SSO login request to HCMUT\_SSO
    \item HCMUT\_SSO returns authentication token with user role
    \item System stores authentication session in Database
    \item System requests user data synchronization from HCMUT\_DATACORE
    \item HCMUT\_DATACORE returns user profiles and academic data
    \item System updates user profiles in Database
    \item Database confirms synchronization completion
    \item System requests resource access from HCMUT\_LIBRARY
    \item HCMUT\_LIBRARY returns learning resources and metadata
    \item System stores resource references in Database
    \item Database confirms resources updated
    \item System sends profile and resource updates to Student
    \item System sends profile and resource updates to Tutor
    \item System sends profile and resource updates to Admin
\end{enumerate}

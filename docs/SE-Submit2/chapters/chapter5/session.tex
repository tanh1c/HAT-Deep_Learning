\noindent \textbf{Description}
\begin{itemize}
  \item \textbf{Purpose:} Model the lifecycle of a tutoring session from booking to completion or cancellation.
  
  \item \textbf{States:} 
    \textit{Scheduled} (Do: Send confirmation, Create calendar event, Wait for session time);
    \textit{Confirmed} (Do: Notify participants, Prepare materials, Check availability);
    \textit{Ongoing} (Do: Start timer, Enable meeting link, Monitor attendance);
    \textit{Completed} (Do: Generate report, Request evaluation, Update progress);
    \textit{Cancelled} (Do: Send cancellation notice, Release time slot, Log reason);
    \textit{Rescheduled} (Do: Find new slot, Update calendar, Notify changes).
  
  \item \textbf{Transitions:}
    Booking $\Rightarrow$ \textit{Scheduled};
    Tutor Accepts $\Rightarrow$ \textit{Confirmed};
    Session Time Reached $\Rightarrow$ \textit{Ongoing};
    Session Ended $\Rightarrow$ \textit{Completed};
    Cancel Request from \textit{Scheduled} or \textit{Confirmed} $\Rightarrow$ \textit{Cancelled};
    Reschedule Request $\Rightarrow$ \textit{Rescheduled};
    New Time Confirmed $\Rightarrow$ \textit{Scheduled};
    Emergency Stop from \textit{Ongoing} $\Rightarrow$ \textit{Cancelled}.
  
  \item \textbf{Start/End:} 
    Initial node $\rightarrow$ \textit{Scheduled}; 
    \textit{Completed} $\rightarrow$ final node;
    \textit{Cancelled} $\rightarrow$ final node.
  
  \item \textbf{Notes:} 
    Only \textit{Confirmed} sessions can transition to \textit{Ongoing}; 
    \textit{Cancelled} sessions release the time slot for other bookings; 
    \textit{Rescheduled} sessions must be re-confirmed before becoming \textit{Ongoing};
    \textit{Completed} sessions trigger evaluation and progress tracking;
    Both \textit{Completed} and \textit{Cancelled} are terminal states in this chart.
\end{itemize}
